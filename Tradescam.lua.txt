function ConfigManager:Load()
    if ENV_TESTING or type(readfile) ~= "function" then return end
    local ok, raw = pcall(readfile, CONFIG_FILE_NAME)
    if not ok or type(raw) ~= "string" then return end
    local okDecode, decoded = pcall(function() return HttpService:JSONDecode(raw) end)
    if okDecode and type(decoded) == "table" then
        for k, v in pairs(decoded) do
            local current = self.Settings[k]
            if current ~= nil then
                if type(v) == "boolean" and type(current) == "boolean" then
                    self.Settings[k] = v
                elseif type(v) == "number" and type(current) == "number" then
                    -- only known numeric settings, always inside their safe range
                    local limits = NUMERIC_LIMITS[k]
                    if limits then
                        self.Settings[k] = math.clamp(v, limits[1], limits[2])
                    end
                end
            end
        end
    end
end

function ConfigManager:Save()
    if ENV_TESTING or type(writefile) ~= "function" then return end
    pcall(function()
        writefile(CONFIG_FILE_NAME, HttpService:JSONEncode(self.Settings))
    end)
end

local Cleanup = { Items = {} }

function Cleanup:Add(item)
    table.insert(self.Items, item)
    return item
end

function Cleanup:Bind(signal, callback)
    return self:Add(signal:Connect(callback))
end

function Cleanup:Clean()
    for _, item in ipairs(self.Items) do
        if typeof(item) == "RBXScriptConnection" then
            pcall(function() item:Disconnect() end)
        end
    end
    self.Items = {}
end

local function create(class, props, parent)
    local inst = Instance.new(class)
    for k, v in pairs(props) do inst[k] = v end
    if parent then inst.Parent = parent end
    return inst
end

local function round(obj, radius)
    return create("UICorner", { CornerRadius = UDim.new(0, radius) }, obj)
end

local function stroke(obj, color, thickness, transparency)
    return create("UIStroke", {
        Color             = color,
        Thickness         = thickness or 1,
        Transparency      = transparency or 0,
        ApplyStrokeMode   = Enum.ApplyStrokeMode.Border,
    }, obj)
end

local function tween(obj, duration, props, style, direction)
    local tw = TweenService:Create(
        obj,
        TweenInfo.new(duration, style or Enum.EasingStyle.Quint, direction or Enum.EasingDirection.Out),
        props
    )
    tw:Play()
    return tw
end

-- ============================================================
-- SPRING SYSTEM
-- ============================================================
local Springs = { List = {} }
local Spring  = {}
Spring.__index = Spring

function Spring.new(initial, stiffness, damping, mass, onUpdate)
    local self = setmetatable({
        Value     = initial,
        Target    = initial,
        Velocity  = 0,
        Stiffness = stiffness,
        Damping   = damping,
        Mass      = mass or 1,
        Awake     = true,
        OnUpdate  = onUpdate,
    }, Spring)
    table.insert(Springs.List, self)
    return self
end

function Spring:Set(target)
    if self.Target ~= target then
        self.Target = target
        self.Awake  = true
    end
end

function Spring:Snap(value)
    self.Value    = value
    self.Target   = value
    self.Velocity = 0
    self.Awake    = true
end

function Spring:Step(dt)
    local steps = math.min(8, math.max(1, math.ceil(dt * 120)))
    local h = dt / steps
    for _ = 1, steps do
        local force = -self.Stiffness * (self.Value - self.Target) - self.Damping * self.Velocity
        self.Velocity = self.Velocity + (force / self.Mass) * h
        self.Value    = self.Value + self.Velocity * h
    end
    if math.abs(self.Value - self.Target) < 0.0005 and math.abs(self.Velocity) < 0.001 then
        self.Value    = self.Target
        self.Velocity = 0
        self.Awake    = false
    end
    if self.OnUpdate then self.OnUpdate(self.Value) end
end

function Springs:Update(dt)
    dt = math.min(dt, 1 / 20)
    for _, spring in ipairs(self.List) do
        if spring.Awake then spring:Step(dt) end
    end
end

local GlowRegistry = {}

local function createGlow(parent, color, spread, intensity, radius, zindex)
    -- prune glows whose host was destroyed (toasts/notices/success bursts)
    for i = #GlowRegistry, 1, -1 do
        local alive = GlowRegistry[i]
        if not alive or not alive.Parent then table.remove(GlowRegistry, i) end
    end
    local layers = ConfigManager.Settings.LiteMode and 4 or 8
    local host = create("Frame", {
        BackgroundTransparency = 1,
        AnchorPoint            = Vector2.new(0.5, 0.5),
        Position               = UDim2.fromScale(0.5, 0.5),
        Size                   = UDim2.new(1, spread * 2, 1, spread * 2),
        ZIndex                 = zindex or 1,
        Visible                = ConfigManager.Settings.Glows,
    }, parent)
    local alpha = (intensity * 1.5) / layers
    for i = 1, layers do
        local inset = ((i - 1) / layers) * spread
        local layer = create("Frame", {
            BackgroundColor3 = color,
            BackgroundTransparency = 1 - alpha,
            BorderSizePixel  = 0,
            AnchorPoint      = Vector2.new(0.5, 0.5),
            Position         = UDim2.fromScale(0.5, 0.5),
            Size             = UDim2.new(1, -inset * 2, 1, -inset * 2),
            ZIndex           = zindex or 1,
        }, host)
        round(layer, (radius or 8) + (spread - inset))
    end
    table.insert(GlowRegistry, host)
    return host
end

local function glassFill(obj)
    obj.BackgroundTransparency = 1
    local fill = create("Frame", {
        Size             = UDim2.fromScale(1, 1),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BorderSizePixel  = 0,
        ZIndex           = 1,
    }, obj)
    local corner = obj:FindFirstChildOfClass("UICorner")
    round(fill, corner and corner.CornerRadius.Offset or 8)
    create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0,    Color3.fromRGB(38, 20, 14)),
            ColorSequenceKeypoint.new(0.55, Color3.fromRGB(22, 10, 8)),
            ColorSequenceKeypoint.new(1,    Color3.fromRGB(14,  6, 5)),
        }),
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 0.06),
            NumberSequenceKeypoint.new(1, 0.2),
        }),
        Rotation = 90,
    }, fill)
    local highlight = create("Frame", {
        Size             = UDim2.new(1, -24, 0, 1),
        Position         = UDim2.fromOffset(12, 0),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BorderSizePixel  = 0,
        ZIndex           = 3,

    create("UIGradient", {
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0,    1),
            NumberSequenceKeypoint.new(0.5, 0.7),
            NumberSequenceKeypoint.new(1,    1),
        }),
    }, highlight)
end

        Color = ColorSequence.new({

    }, s)
    return s
end

local function drawIcon(parent, name, size, color, pos)
    local u    = size / 24
    local root = create("Frame", {
        BackgroundTransparency = 1,
        Size                   = UDim2.fromOffset(size, size),
        Position               = pos or UDim2.new(),
        AnchorPoint            = Vector2.new(0.5, 0.5),
    }, parent)
    local parts = {}

    local function seg(x1, y1, x2, y2, t)
        local dx, dy = x2 - x1, y2 - y1
        local len    = math.sqrt(dx * dx + dy * dy)
        local th     = math.max((t or 1.8) * u, 1)
        local f = create("Frame", {
            BackgroundColor3 = color,
            BorderSizePixel  = 0,
            AnchorPoint      = Vector2.new(0.5, 0.5),
            Position         = UDim2.fromOffset((x1 + x2) / 2 * u, (y1 + y2) / 2 * u),
            Size             = UDim2.fromOffset(len * u + th, th),
            Rotation         = math.deg(math.atan2(dy, dx)),
        }, root)
        round(f, th / 2)
        table.insert(parts, { f, "fill" })
    end

    local function ring(cx, cy, w, h, t, r)
        local f = create("Frame", {
            BackgroundTransparency = 1,
            AnchorPoint            = Vector2.new(0.5, 0.5),
            Position               = UDim2.fromOffset(cx * u, cy * u),
            Size                   = UDim2.fromOffset(w * u, h * u),
        }, root)
        round(f, r * u)
        local s = stroke(f, color, math.max(t * u, 1))
        table.insert(parts, { s, "stroke" })
    end

    local function dot(cx, cy, r)
        local f = create("Frame", {
            BackgroundColor3 = color,
            BorderSizePixel  = 0,
            AnchorPoint      = Vector2.new(0.5, 0.5),
            Position         = UDim2.fromOffset(cx * u, cy * u),
            Size             = UDim2.fromOffset(r * 2 * u, r * 2 * u),
        }, root)
        round(f, r * u)
        table.insert(parts, { f, "fill" })
    end

    local function arc(cx, cy, r, a0, a1, t, n)
        n = n or 8
        local px, py
        for i = 0, n do
            local a = math.rad(a0 + (a1 - a0) * i / n)
            local x, y = cx + math.cos(a) * r, cy + math.sin(a) * r
            if px then seg(px, py, x, y, t) end
            px, py = x, y
        end
    end

    local function poly(points, t)
        for i = 1, #points - 1 do
            seg(points[i][1], points[i][2], points[i + 1][1], points[i + 1][2], t)
        end
    end

    if name == "shield" then

    elseif name == "settings" then
        ring(12,12, 9,9, 1.8, 4.5)
        for i = 0, 7 do
            local a = math.rad(i * 45)
            seg(12+math.cos(a)*6.8, 12+math.sin(a)*6.8, 12+math.cos(a)*10, 12+math.sin(a)*10, 2.4)
        end
    elseif name == "check" then
        seg(5,12.5, 10,17.5, 2.2); seg(10,17.5, 19.5,7, 2.2)
    elseif name == "copy" then
        ring(14.5,14.5, 12,12, 1.6, 2.5); ring(9.5,9.5, 12,12, 1.6, 2.5)
    elseif name == "user" then
        ring(12,8, 7,7, 1.8, 3.5)
        arc(12,21.5, 8, 180, 360, 1.8, 10)
        seg(4,21.5, 20,21.5, 1.8)
    elseif name == "terminal" then
        ring(12,12, 20,16, 1.6, 3)
        seg(7,9.5, 11,12.5, 1.8); seg(11,12.5, 7,15.5, 1.8)
        seg(13.5,16, 17,16, 1.8)
    elseif name == "lock" then
        ring(12,16, 14,9, 1.8, 2.5)
        arc(12,10, 4.5, 180, 360, 1.8, 8)
        seg(7.5,10, 7.5,11.8, 1.8); seg(16.5,10, 16.5,11.8, 1.8)
        dot(12,16, 1.3)
    elseif name == "close" then
        seg(6,6, 18,18, 2); seg(18,6, 6,18, 2)

local function applyRipple(button)
    button.ClipsDescendants = true
    Cleanup:Bind(button.InputBegan, function(input)
        local kind = input.UserInputType
        if kind ~= Enum.UserInputType.MouseButton1 and kind ~= Enum.UserInputType.Touch then return end
        local point
        if kind == Enum.UserInputType.Touch then
            point = Vector2.new(input.Position.X, input.Position.Y) + GuiService:GetGuiInset()
        else
            point = UserInputService:GetMouseLocation()
        end
        local size = button.AbsoluteSize
        local rel  = point - button.AbsolutePosition
        local ripple = create("Frame", {
            Size                   = UDim2.fromOffset(0, 0),
            Position               = UDim2.fromScale(rel.X / math.max(size.X,1), rel.Y / math.max(size.Y,1)),
            AnchorPoint            = Vector2.new(0.5, 0.5),
            BackgroundColor3       = Color3.new(1, 1, 1),
            BackgroundTransparency = 0.55,
            BorderSizePixel        = 0,
            ZIndex                 = button.ZIndex + 5,
        }, button)
        round(ripple, 999)
        local target = math.max(size.X, size.Y) * 2.4
        local tw = TweenService:Create(ripple, TweenInfo.new(0.65, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
            Size                   = UDim2.fromOffset(target, target),
            BackgroundTransparency = 1,
        })
        tw:Play()
        tw.Completed:Connect(function() ripple:Destroy() end)
    end)
end

local function attachInteraction(button, target)
    target = target or button
    local scale  = create("UIScale", { Scale = 1 }, target)
    local spring = Spring.new(1, 340, 15, 1, function(v) scale.Scale = v end)
    local hovering, pressed = false, false
    local function refresh()
        spring:Set(pressed and 0.96 or (hovering and 1.04 or 1))
    end
    local function isPress(input)
        return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
    end
    Cleanup:Bind(button.MouseEnter,  function() hovering = true;  refresh() end)
    Cleanup:Bind(button.MouseLeave,  function() hovering = false; pressed = false; refresh() end)
    Cleanup:Bind(button.InputBegan,  function(input) if isPress(input) then pressed = true;  refresh() end end)
    Cleanup:Bind(button.InputEnded,  function(input) if isPress(input) then pressed = false; refresh() end end)
    return spring
end

local function FireToast(gui, title, kind, detail, iconName)
    local color = Theme.Accent
    if kind == "Success" then color = Theme.Success
    elseif kind == "Error"   then color = Theme.Error
    elseif kind == "Warning" then color = Theme.Warning end
    local height = detail and 70 or 46
    local guiW   = (gui.AbsoluteSize.X > 0) and gui.AbsoluteSize.X or 800
    local holder = create("Frame", {
        Size                   = UDim2.fromOffset(math.clamp(guiW - 48, 240, 320), height),
        AnchorPoint            = Vector2.new(1, 1),
        BackgroundTransparency = 1,
        Position               = UDim2.new(1, 380, 1, -20),
        ZIndex                 = 100,
    }, gui)
    createGlow(holder, color, 16, 0.4, 8, 100)
    local card = create("CanvasGroup", { Size = UDim2.fromScale(1,1), GroupTransparency=0, BorderSizePixel=0, ZIndex=101 }, holder)
    round(card, 8)
    glassFill(card)
    glassStroke(card)
    local bar = create("Frame", {
        Size             = UDim2.new(0, 3, 1, -18),
        Position         = UDim2.fromOffset(9, 9),
        BackgroundColor3 = color,
        BorderSizePixel  = 0,
        ZIndex           = 4,
    }, card)
    round(bar, 2)
    local icon = iconName or (kind=="Error" and "close" or (kind=="Success" and "check" or "shield"))
    drawIcon(card, icon, 18, color, UDim2.fromOffset(34, detail and 24 or height/2))
    makeLabel(card, {
        Text         = title,
        TextSize     = 12,
        Font         = Enum.Font.GothamBold,
        Position     = UDim2.fromOffset(54, detail and 8 or 0),
        Size         = UDim2.new(1, -64, 0, detail and 20 or height),
        TextTruncate = Enum.TextTruncate.AtEnd,
    })
    if detail then
        makeLabel(card, {
            Text                = detail,

local function edgeFlash(parent, color, size, pulses, z)
    local host  = create("Frame", { Size = UDim2.fromScale(1,1), BackgroundTransparency=1, ZIndex=z }, parent)
    local edges = {}
    local defs  = {
        { UDim2.new(1,0,0,size), UDim2.fromOffset(0,0),        90  },
        { UDim2.new(1,0,0,size), UDim2.new(0,0,1,-size),       270 },
        { UDim2.new(0,size,1,0), UDim2.fromOffset(0,0),        0   },
        { UDim2.new(0,size,1,0), UDim2.new(1,-size,0,0),       180 },
    }
    for _, d in ipairs(defs) do
        local e = create("Frame", {
            Size                   = d[1],
            Position               = d[2],
            BackgroundColor3       = color,
            BackgroundTransparency = 1,
            BorderSizePixel        = 0,
            ZIndex                 = z,
        }, host)
        create("UIGradient", {
            Rotation     = d[3],
            Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0, 0.1),
                NumberSequenceKeypoint.new(1, 1),
            }),
        }, e)
        table.insert(edges, e)
    end
    task.spawn(function()
        for _ = 1, pulses do
            for _, e in ipairs(edges) do tween(e, 0.12, { BackgroundTransparency = 0 }, Enum.EasingStyle.Quad) end
            task.wait(0.14)
            for _, e in ipairs(edges) do tween(e, 0.3,  { BackgroundTransparency = 1 }, Enum.EasingStyle.Quad) end
            task.wait(0.32)
        end
        host:Destroy()
    end)
    return host
end

    end)

local function createBatParticles(parent, count, zindex)
    local field = create("Frame", {
        Size                   = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1,
        ZIndex                 = zindex,
        Visible                = ConfigManager.Settings.Particles,
    }, parent)
    local list = {}
    for _ = 1, count do
        local sz = 14 + math.random(0, 8)
        local holder = create("Frame", {
            BackgroundTransparency = 1,
            Size                   = UDim2.fromOffset(sz, sz),
            AnchorPoint            = Vector2.new(0.5, 0.5),
            Position               = UDim2.fromScale(math.random(), math.random()),
            ZIndex                 = zindex,
        }, field)
        drawIcon(holder, "bat", sz, Color3.fromRGB(50, 20, 20), UDim2.fromScale(0.5, 0.5))
        -- cache the icon frames: GetDescendants() every frame was GC churn
        local iconParts = {}
        for _, child in ipairs(holder:GetDescendants()) do
            if child:IsA("Frame") then table.insert(iconParts, child) end
        end
        local dirRight = math.random() < 0.5
        table.insert(list, {
            holder    = holder,
            x         = math.random(),
            y         = math.random(),
            speed     = 0.012 + math.random() * 0.022,
            phase     = math.random() * math.pi * 2,
            amp       = 0.012 + math.random() * 0.02,
            freq      = 0.6 + math.random() * 1.0,
            dirRight  = dirRight,
            base      = 0.55 + math.random() * 0.3,
            depth     = 0.3 + math.random() * 0.7,
            flipTimer = math.random() * 5,
            parts     = iconParts,
        })
    end
    return field, list
end

local function updateBatParticles(list, t, dt, norm)
    for _, p in ipairs(list) do
        p.flipTimer = p.flipTimer - dt
        if p.flipTimer <= 0 then
            if math.random() < 0.2 then p.dirRight = not p.dirRight end
            p.flipTimer = 3 + math.random() * 4
        end
        local dir = p.dirRight and 1 or -1
        p.x = p.x + p.speed * dir * dt * 60 / 60
        if p.x > 1.06 then p.x = -0.06 elseif p.x < -0.06 then p.x = 1.06 end
        local x = p.x + math.sin(t * p.freq + p.phase) * p.amp - norm.X * p.depth * 0.03
        local y = p.y - norm.Y * p.depth * 0.02
        y = y + math.sin(t * p.freq * 1.3 + p.phase) * 0.015
        if y < -0.05 then y = 1.05 end
        if y > 1.05  then y = -0.05 end
        p.y = y
        p.holder.Position                  = UDim2.fromScale(x, y)
        p.holder.Rotation                  = dir < 0 and 180 or 0
        local alpha = math.clamp(p.base + math.sin(t * p.freq * 2 + p.phase) * 0.15, 0, 1)
        for _, part in ipairs(p.parts) do
            part.BackgroundTransparency = alpha
        end
    end
end

local function truncate(text, limit)
    text = tostring(text or "")
    limit = tonumber(limit) or 200
    if #text <= limit then return text end
    if limit <= 3 then return string.sub(text, 1, limit) end
    return string.sub(text, 1, limit - 3) .. "..."
end

    end

    end

    if keyboard then table.insert(inputs, "Keyboard") end

    if touch    then table.insert(inputs, "Touch")    end

        device     = device,
        inputs     = #inputs > 0 and table.concat(inputs, ", ") or "Unknown",
